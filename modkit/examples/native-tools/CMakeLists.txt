cmake_minimum_required(VERSION 3.25)
project(SimpsonsMod LANGUAGES CXX)
if(NOT DEFINED MODKIT_SDK_DIR)
    message(FATAL_ERROR "Set MODKIT_SDK_DIR to the Simpsons Modkit directory")
endif()
set(SIMPSONS_MODKIT_RUNTIME OFF CACHE BOOL "SDK headers only" FORCE)
add_subdirectory("${MODKIT_SDK_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
add_library(simpsons_mod MODULE main.cpp)
target_link_libraries(simpsons_mod PRIVATE simpsons::modkit_sdk)
set_target_properties(simpsons_mod PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
